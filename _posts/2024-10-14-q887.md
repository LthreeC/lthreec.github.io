给你 k 枚相同的鸡蛋，并可以使用一栋从第 1 层到第 n 层共有 n 层楼的建筑。

已知存在楼层 f ，满足 0 <= f <= n ，任何从 高于 f 的楼层落下的鸡蛋都会碎，从 f 楼层或比它低的楼层落下的鸡蛋都不会破。

每次操作，你可以取一枚没有碎的鸡蛋并把它从任一楼层 x 扔下（满足 1 <= x <= n）。如果鸡蛋碎了，你就不能再次使用它。如果某枚鸡蛋扔下后没有摔碎，则可以在之后的操作中 重复使用 这枚鸡蛋。

请你计算并返回要确定 f 确切的值 的 最小操作次数 是多少？

 
示例 1：

输入：k = 1, n = 2
输出：2
解释：
鸡蛋从 1 楼掉落。如果它碎了，肯定能得出 f = 0 。 
否则，鸡蛋从 2 楼掉落。如果它碎了，肯定能得出 f = 1 。 
如果它没碎，那么肯定能得出 f = 2 。 
因此，在最坏的情况下我们需要移动 2 次以确定 f 是多少。 
示例 2：

输入：k = 2, n = 6
输出：3
示例 3：

输入：k = 3, n = 14
输出：4
 

```cpp
class Solution {
public:
    int superEggDrop(int k, int n) {
        int f[105][10005];
        memset(f, 0, sizeof(f));
        for (int j = 1; j <= n; ++j) {
            f[1][j] = j;
        }
        for (int i = 1; i <= k; ++i) {
            f[i][1] = 1;
        }
        for (int i = 2; i <= k; ++i) {
            for (int j = 2; j <= n; ++j) {
                // cout << " j = " << j << "\n";
                int l = 1, r = j - 1, lo = l, hi = r;
                while (l <= r) {
                    int mid = (l + r) / 2;
                    // 居然是利用单调性
                    if (f[i - 1][mid - 1] <= f[i][j - mid]) {
                        lo = mid;
                        l = mid + 1;
                    }
                    else {
                        r = mid - 1;
                    }
                }
                hi = lo + 1;

                // int lo = 1, hi = j;
                // while (lo + 1 < hi) {
                //     int x = (lo + hi) / 2;
                //     int t1 = f[i - 1][x - 1];
                //     int t2 = f[i][j - x];

                //     if (t1 < t2) {
                //         lo = x;
                //     }
                //     else if (t1 > t2) {
                //         hi = x;
                //     }
                //     else {
                //         lo = hi = x;
                //     }
                // }

                // cout << " j= " << j << " " << lo << " " << hi << "\n";

                f[i][j] = 1 + min(max(f[i - 1][lo - 1], f[i][j - lo]), max(f[i - 1][hi - 1], f[i][j - hi]));
            }
        }

        // for (int i = 1; i <= k; ++i) {
        //     for (int j = 1; j <= n; ++j) {
        //         cout << f[i][j] << " ";
        //     }
        //     cout << "\n";
        // }

        return f[k][n];
    }
};
```
